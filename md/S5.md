```sql
CREATE TABLE program_outcomes (

    pso_code VARCHAR(20) NOT NULL, -- PSO1, PSO2
    description TEXT NOT NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    UNIQUE(program_id, pso_code)
);

CREATE TABLE course_learning_outcomes (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    course_id UUID NOT NULL REFERENCES courses(id),
    clo_code VARCHAR(20) NOT NULL, -- CLO1, CLO2
    description TEXT NOT NULL,
    bloom_level VARCHAR(50),
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    UNIQUE(course_id, clo_code)
);

CREATE TABLE clo_po_mapping (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    clo_id UUID NOT NULL REFERENCES course_learning_outcomes(id),
    po_id UUID NOT NULL REFERENCES program_outcomes(id),
    mapping_strength VARCHAR(20) NOT NULL DEFAULT 'medium', -- weak | medium | strong
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    UNIQUE(clo_id, po_id)
);

CREATE TABLE clo_pso_mapping (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    clo_id UUID NOT NULL REFERENCES course_learning_outcomes(id),
    pso_id UUID NOT NULL REFERENCES program_specific_outcomes(id),
    mapping_strength VARCHAR(20) NOT NULL DEFAULT 'medium',
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    UNIQUE(clo_id, pso_id)
);

CREATE TABLE assessment_clo_mapping (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    assessment_component_id UUID NOT NULL REFERENCES assessment_components(id),
    clo_id UUID NOT NULL REFERENCES course_learning_outcomes(id),
    weight DECIMAL(5, 2) NOT NULL, -- % of assessment measuring this CLO
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
```

## Task 2 — OBE Attainment Calculation

### Service
```csharp
public interface IOBEAttainmentService
{
    Task<CLOAttainment> CalculateCLOAttainmentAsync(Guid courseId, Guid cloId, Guid semesterId, CancellationToken ct = default);
    Task<POAttainment> CalculatePOAttainmentAsync(Guid programId, Guid poId, Guid semesterId, CancellationToken ct = default);
    Task<ProgramOBEReport> GenerateProgramReportAsync(Guid programId, Guid semesterId, CancellationToken ct = default);
}

public class CLOAttainment
{
    public Guid CLOId { get; set; }
    public string CLOCode { get; set; }
    public decimal AttainmentPercentage { get; set; } // 0-100
    public int TotalStudents { get; set; }
    public int StudentsMeeting70Threshold { get; set; }
    public List<AssessmentContribution> AssessmentBreakdown { get; set; }
}

public class POAttainment
{
    public Guid POId { get; set; }
    public string POCode { get; set; }
    public decimal AttainmentPercentage { get; set; }
    public List<CLOMappingContribution> MappedCLOs { get; set; } // how each CLO contributes
}
```

### Calculation Logic
1. **CLO Attainment** = weighted average of assessments mapped to that CLO
   - For each assessment component: students' average score in that component
   - Weight by `assessment_clo_mapping.weight`
   - Sum across all assessments for that CLO
   - Result: % of students achieving >= 70% in that CLO

2. **PO Attainment** = weighted average of mapped CLOs
   - For each CLO mapped to PO: use CLO attainment percentage
   - Weight by `clo_po_mapping.mapping_strength` (weak=0.33, medium=0.66, strong=1.0)
   - Sum across all CLOs
   - Result: aggregate PO attainment across semester

3. **Program-level**: average of all POs for semester

## Task 4 — CIS Scanner (Azure Document Intelligence)

### Overview
CIS = Course Information Sheet (standard at Pakistani universities)
Goal: Auto-extract CLOs, PLOs, credit hours, assessment methods from faculty-uploaded PDFs

### Setup
- Train Azure Document Intelligence (formerly Form Recognizer) on SSUET's CIS template
- Upload 5-10 sample CIS PDFs
- Mark fields: Course Title, Course Code, Credit Hours, CLO1, CLO2, ..., PLO1, ..., Assessment
- Model learns the layout

### Database Tables
```sql
CREATE TABLE cis_uploads (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    course_id UUID NOT NULL REFERENCES courses(id),
    faculty_id UUID NOT NULL REFERENCES users(id),
    cis_file_path TEXT NOT NULL,
    upload_date TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    processing_status VARCHAR(20) DEFAULT 'pending', -- pending | processing | completed | failed
    extracted_data JSONB,
    human_review_status VARCHAR(20) DEFAULT 'pending', -- pending | approved | rejected
    reviewed_by UUID REFERENCES users(id),
    reviewed_at TIMESTAMPTZ,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
```

### Extraction Workflow

1. **Faculty uploads CIS** (PDF or image)
   - Endpoint: `POST /api/courses/{courseId}/cis/upload`
   - File stored via `IStorageProvider`
   - Create `cis_uploads` record (status=pending)

2. **Background job: Run Azure Document Intelligence**
   ```csharp
   public class CISExtractionService
   {
       public async Task ProcessCISAsync(Guid cisUploadId, CancellationToken ct = default)
       {
           var cisUpload = await _db.CISUploads.FindAsync(cisUploadId);
           var fileStream = await _storage.DownloadAsync(cisUpload.CISFilePath, ct);
           
           var client = new DocumentAnalysisClient(...azure config...);
           var operation = await client.AnalyzeDocumentAsync(
               WaitUntil.Completed,
               "cis-model", // trained model
               fileStream,
               ct
           );
           
           var result = operation.Value;
           
           // Extract fields
           var extracted = new Dictionary<string, string>
           {
               { "courseTitle", result.Fields["courseTitle"].Value },
               { "courseCode", result.Fields["courseCode"].Value },
               { "creditHours", result.Fields["creditHours"].Value },
               { "clo1", result.Fields["clo1"].Value },
               { "clo2", result.Fields["clo2"].Value },
               // ... more CLOs
           };
           
           cisUpload.ExtractedData = JsonConvert.SerializeObject(extracted);
           cisUpload.ProcessingStatus = "completed";
           await _db.SaveChangesAsync(ct);
       }
   }
   ```

3. **Human Review** (Faculty side)
   - Faculty views extracted data
   - Can edit / correct any field
   - Confirms accuracy via checkbox
   - Status → approved

4. **On Approval: Auto-populate OBE**
   - Create/update `course_learning_outcomes` from extracted CLOs
   - Create/update `courses.title`, `courses.code`, `courses.credits`
   - Prepopulate `assessment_clo_mapping` from extracted assessment methods

### Integration with Moodle
- On CIS approval: sync CLOs to Moodle course outline
- Moodle activity descriptions can reference mapped CLOs
