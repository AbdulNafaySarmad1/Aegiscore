#### THREAT 1: SQL Injection

**Current Risk:**
```csharp
// ❌ VULNERABLE
var query = $"SELECT * FROM users WHERE email = '{userEmail}'";
var result = await _db.Users.FromSqlInterpolated(query).ToListAsync();
```

**Fix:**
```csharp
// ✅ SAFE
var result = await _db.Users
    .Where(u => u.Email == userEmail)
    .ToListAsync();

// OR parameterized query
var result = await _db.Users
    .FromSqlInterpolated($"SELECT * FROM users WHERE email = {userEmail}")
    .ToListAsync();
```

**Audit Steps:**
1. Scan all repositories for string interpolation in SQL
2. Replace with Entity Framework LINQ or parameterized queries
3. Test with SQL injection payloads:
   ```
   ' OR '1'='1
   ' OR 1=1; DROP TABLE users; --
   admin'; DELETE FROM students; --
   ```

#### THREAT 4: Authentication Bypass

**Current Risk:**
```csharp
// ❌ VULNERABLE
public async Task<User> LoginAsync(string email, string password)
{
    var user = await _db.Users.FirstOrDefaultAsync(u => u.Email == email);
    if (user.PasswordHash == password) // Plain text comparison!
        return user;
    return null;
}
```

**Fix:**
```csharp
// ✅ SAFE
public async Task<User> LoginAsync(string email, string password)
{
    var user = await _db.Users.FirstOrDefaultAsync(u => u.Email == email);
    
    if (user == null)
        return null; // No such user
    
    // Use bcrypt for comparison
    bool isValid = BCrypt.Net.BCrypt.Verify(password, user.PasswordHash);
    
    if (!isValid)
    {
        // Log failed attempt
        await _auditService.LogLoginFailureAsync(email);
        
        // Rate limit (5 failures = lockout)
        if (await _auditService.GetFailedAttemptsAsync(email) >= 5)
            return null; // Account locked
        
        return null;
    }
    
    // Successful login
    await _auditService.LogLoginSuccessAsync(email);
    await _auditService.ResetFailedAttemptsAsync(email);
    return user;
}
```

**Audit Steps:**
1. Check all passwords are hashed (bcrypt, not MD5/SHA1)
2. Implement rate limiting (5 failed attempts = lockout)
3. Log all authentication attempts
4. Test with:
   - Weak passwords (should be rejected)
   - Brute force attempts (should be rate-limited)
   - Default credentials (should not exist)

#### THREAT 6: Weak Authorization (RBAC Bypass)

**Current Risk:**
```csharp
// ❌ VULNERABLE
public async Task<List<Student>> GetStudentsAsync(Guid departmentId)
{
    // No check if user is HOD of this department!
    return await _db.Students
        .Where(s => s.DepartmentId == departmentId)
        .ToListAsync();

```

**Fix:**
```csharp
// ✅ SAFE
public async Task<List<Student>> GetStudentsAsync(Guid departmentId)
{
    var currentUser = await _userService.GetCurrentUserAsync();
    
    // Check authorization
    if (currentUser.Role == "VC")
    {
        // VC can see all students
    }
    else if (currentUser.Role == "HOD")
    {
        // HOD can only see their own department
        if (currentUser.DepartmentId != departmentId)
            throw new UnauthorizedAccessException("Cannot access other departments");
    }
    else if (currentUser.Role == "Faculty")
    {
        // Faculty can only see students in their courses
        throw new ForbiddenAccessException("Faculty cannot view all students");
    }
    else
    {
        throw new UnauthorizedAccessException("Invalid role");
    }
    
    return await _db.Students
        .Where(s => s.DepartmentId == departmentId)
        .ToListAsync();
}
```

**Audit Steps:**
1. Check every API endpoint for authorization
2. Verify HOD can't access other departments
3. Verify Faculty can't see all grades
4. Verify Students can't see each other's data
5. Test with:
   - HOD token accessing different department
   - Faculty token accessing student data
   - Student token accessing admin data

### 2.3 Supabase License Database

**Schema:**

```sql
-- Universities table
CREATE TABLE universities (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    name VARCHAR(255) NOT NULL,
    website VARCHAR(255),
    email VARCHAR(255),
    admin_id UUID NOT NULL,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(email)
);

-- Licenses table (RLS enabled)
CREATE TABLE licenses (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    university_id UUID REFERENCES universities(id) ON DELETE CASCADE,
    license_key VARCHAR(255) NOT NULL UNIQUE,
    
    -- License validity
    issued_at TIMESTAMPTZ DEFAULT NOW(),
    paid_until DATE NOT NULL, -- Expiry date
    status ENUM('active', 'suspended', 'expired') DEFAULT 'active',
    
    -- Payment tracking
    payment_date DATE,
    amount_paid DECIMAL(10,2),
    payment_method VARCHAR(50), -- 'bank_transfer', 'card', 'crypto'
    invoice_id VARCHAR(255),
    
    -- Usage
    last_heartbeat TIMESTAMPTZ,
    daily_check_at TIMESTAMPTZ,
    
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Audit log (who changed what)
CREATE TABLE license_audits (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    license_id UUID REFERENCES licenses(id),
    admin_id UUID REFERENCES auth.users(id),
    action VARCHAR(50), -- 'created', 'suspended', 'reactivated', 'expired'
    old_status VARCHAR(50),
    new_status VARCHAR(50),
    reason TEXT,
    timestamp TIMESTAMPTZ DEFAULT NOW()
);

-- Enable Row Level Security (RLS)
ALTER TABLE licenses ENABLE ROW LEVEL SECURITY;
ALTER TABLE license_audits ENABLE ROW LEVEL SECURITY;

-- RLS Policy: Universities can only see their own licenses
CREATE POLICY "Universities see own licenses"
ON licenses FOR ALL
USING (
    -- University ID must match current user's university
    university_id IN (
        SELECT id FROM universities 
        WHERE admin_id = auth.uid()
    )
);

-- RLS Policy: Only admins can see audit logs
CREATE POLICY "Admins see all audits"
ON license_audits FOR ALL
USING (
    -- Only Abdul (admin) can see
    auth.uid() = 'abdul-user-id'
);
```

### 2.4 License Checker Service (Node.js)

**File:** `/home/ubuntu/aegiscore-license-checker/src/license-checker.js`

```javascript
const { createClient } = require('@supabase/supabase-js');
const axios = require('axios');

const supabase = createClient(
    process.env.SUPABASE_URL,
    process.env.SUPABASE_KEY
);

// Daily check job (8 AM Pakistan time)
async function dailyLicenseCheck() {
    console.log(`[${new Date().toISOString()}] Starting daily license check`);
    
    try {
        // Step 1: Fetch all licenses from Supabase
        const { data: licenses, error } = await supabase
            .from('licenses')
            .select('*, universities(*)')
            .order('updated_at', { ascending: true });
        
        if (error) throw error;
        
        let updated = 0;
        let expired = 0;
        
        // Step 2: Check each license
        for (const license of licenses) {
            const today = new Date().toISOString().split('T')[0];
            const paidUntil = license.paid_until;
            
            if (paidUntil < today && license.status === 'active') {
                // License expired! Suspend it
                const { error: updateError } = await supabase
                    .from('licenses')
                    .update({
                        status: 'expired',
                        updated_at: new Date().toISOString()
                    })
                    .eq('id', license.id);
                
                if (updateError) {
                    console.error(`Failed to expire license ${license.id}:`, updateError);
                } else {
                    console.log(`✅ Expired license: ${license.university_id}`);
                    expired++;
                    
                    // Notify the university
                    await notifyUniversity(license.university_id, 'expired');
                }
            } else if (paidUntil >= today && license.status === 'expired') {
                // License was paid! Reactivate it
                const { error: updateError } = await supabase
                    .from('licenses')
                    .update({
                        status: 'active',
                        updated_at: new Date().toISOString()
                    })
                    .eq('id', license.id);
                
                if (updateError) {
                    console.error(`Failed to reactivate license ${license.id}:`, updateError);
                } else {
                    console.log(`✅ Reactivated license: ${license.university_id}`);
                    updated++;
                    
                    // Notify the university
                    await notifyUniversity(license.university_id, 'reactivated');
                }
            }
            
            // Step 3: Update heartbeat
            await supabase
                .from('licenses')
                .update({
                    last_heartbeat: new Date().toISOString(),
                    daily_check_at: new Date().toISOString()
                })
                .eq('id', license.id);
        }
        
        console.log(`[SUMMARY] Checked ${licenses.length} licenses | Expired: ${expired} | Updated: ${updated}`);
    } catch (error) {
        console.error('Daily license check failed:', error);
    }
}

// Notify university via email/webhook
async function notifyUniversity(universityId, action) {
    try {
        const { data: university } = await supabase
            .from('universities')
            .select('email')
            .eq('id', universityId)
            .single();
        
        if (action === 'expired') {
            // Send "license expired" email
            await axios.post(`${process.env.AEGISCORE_API_URL}/api/notifications/email`, {
                to: university.email,
                subject: '⚠️ Your AegisCore License Has Expired',
                body: `Your license expired on ${new Date().toISOString().split('T')[0]}. 
                       Your AegisCore ERP system is now locked.
                       Contact us to renew your license.`
            }, {
                headers: { 'Authorization': `Bearer ${process.env.AEGISCORE_API_KEY}` }
            });
        } else if (action === 'reactivated') {
            // Send "license reactivated" email
            await axios.post(`${process.env.AEGISCORE_API_URL}/api/notifications/email`, {
                to: university.email,
                subject: '✅ Your AegisCore License Has Been Reactivated',
                body: `Thank you for your payment. Your license has been reactivated.
                       Your AegisCore ERP system is now fully operational.`
            }, {
                headers: { 'Authorization': `Bearer ${process.env.AEGISCORE_API_KEY}` }
            });
        }
    } catch (error) {
        console.error('Failed to notify university:', error);
    }
}

// Schedule daily check at 8 AM Pakistan time
const cron = require('node-cron');
const schedule = '0 8 * * *'; // 8 AM every day (Cron format)

cron.schedule(schedule, () => {
    dailyLicenseCheck();
}, {
    timezone: "Asia/Karachi"
});

console.log('License checker started. Scheduled for 8 AM Pakistan time daily.');
```

### 2.5 License Validation in AegisCore Backend

**Every request must validate license:**

```csharp
public class LicenseValidationMiddleware
{
    private readonly RequestDelegate _next;
    private readonly ISupabaseClient _supabase;
    
    public LicenseValidationMiddleware(RequestDelegate next, ISupabaseClient supabase)
    {
        _next = next;
        _supabase = supabase;
    }
    
    public async Task InvokeAsync(HttpContext context)
    {
        // Extract university ID from request
        var universityId = context.Request.Headers["X-University-ID"].ToString();
        
        if (string.IsNullOrEmpty(universityId))
        {
            context.Response.StatusCode = 400;
            await context.Response.WriteAsync("Missing X-University-ID header");
            return;
        }
        
        // Fetch license from Supabase
        var { data: license, error } = await _supabase
            .From("licenses")
            .Where(l => l.university_id == universityId)
            .Single();
        
        if (license == null || license.status != "active")
        {
            context.Response.StatusCode = 403;
            await context.Response.WriteAsync("License expired or suspended");
            return;
        }
        
        // License valid, continue
        await _next(context);
    }
}
```

```csharp
public class LicenseGenerationService
{
    public async Task<string> GenerateLicenseAsync(Guid universityId, int years = 1)
    {
        // Generate random license key (256-bit)
        var licenseKey = GenerateRandomKey(32); // 64 hex chars
        
        // Calculate expiry (1 year from today)
        var paidUntil = DateTime.Today.AddYears(years);
        
        // Create in Supabase
        var { data: license, error } = await _supabase
            .From("licenses")
            .Insert(new {
                university_id = universityId,
                license_key = licenseKey,
                paid_until = paidUntil,
                status = "active",
                issued_at = DateTime.UtcNow
            })
            .Select()
            .Single();
        
        return licenseKey;
    }
    
    private string GenerateRandomKey(int bytes)
    {
        using (var rng = new System.Security.Cryptography.RNGCryptoServiceProvider())
        {
            var tokenData = new byte[bytes];
            rng.GetBytes(tokenData);
            return Convert.ToHexString(tokenData);
        }
    }
}
```

```csharp
public class LicenseService
{
    public async Task<(bool IsValid, string Message)> ValidateLicenseAsync(string licenseKey)
    {
        // Query Supabase
        var { data: license } = await _supabase
            .From("licenses")
            .Where(l => l.license_key == licenseKey)
            .Single();
        
        if (license == null)
            return (false, "License key not found");
        
        if (license.status == "suspended")
            return (false, "License suspended by admin");
        
        if (license.paid_until < DateTime.Today)
            return (false, $"License expired on {license.paid_until}");
        
        // Valid!
        return (true, "License is valid");
    }
}
```
